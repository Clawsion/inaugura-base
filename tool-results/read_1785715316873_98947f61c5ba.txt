    73→function simpleToGenerateInput(sf: SimpleForgeValues): import("@/lib/schema/inaugura-pack").GenerateInput {
    74→  const projectTypeMap: Record<string, import("@/lib/schema/inaugura-pack").GenerateInput["project_type"]> = {
    75→    landing: "portfolio",
    76→    saas: "saas",
    77→    ecommerce: "ecommerce",
    78→    portfolio: "portfolio",
    79→    dashboard: "saas",
    80→    blog: "corporate",
    81→    marketplace: "ecommerce",
    82→    other: "other",
    83→  };
    84→
    85→  // Mapear integrações simples para IDs do catálogo
    86→  const integrationMap: Record<string, string> = {
    87→    auth: "auth-clerk",
    88→    payments: "stripe",
    89→    database: "supabase",
    90→    email: "resend",
    91→    analytics: "posthog",
    92→    cms: "cms-sanity",
    93→    ai: "posthog", // placeholder
    94→    i18n: "i18n",
    95→  };
    96→
    97→  return {
    98→    locale: sf.idioma,
    99→    brief: sf.briefing,
   100→    project_type: projectTypeMap[sf.projectType] || "other",
   101→    references: sf.references.filter((r) => r.trim()).map((url) => ({ url, role: "visual_anchor" as const })),
   102→    features: sf.integrations.map((i) => integrationMap[i] ?? i),
   103→    sections_lock: [],
   104→    effects_lock: sf.animations ? ["reveal-scroll", "smooth-scroll"] : [],
   105→    visual: {
   106→      locks: { aesthetic: sf.aesthetic, mood: sf.mood.join(","), palette: sf.palette },
   107→    },
   108→    execution: {
   109→      mode: "auto",
   110→      cost_profile: sf.level === "enterprise" ? "max" : sf.level === "production" ? "balanced" : "free_open",
   111→      host_preference: "opencode",
   112→    },
   113→    locks: {
   114→      skills: [],
   115→      mcps: [],
   116→      integrations: sf.integrations.map((i) => integrationMap[i] ?? i),
   117→    },
   118→    level: sf.level === "enterprise" ? "awwwards" : sf.level === "production" ? "pro" : "lite",
   119→    options: {
   120→      polish_design: false,
   121→      include_opencode_json: true,
   122→      include_zip_markdown: true,
   123→    },
   124→  };
   125→}
