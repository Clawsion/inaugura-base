   218→  const toggleArray = useCallback((key: "references" | "mood" | "integrations", item: string) => {
   219→    const arr = value[key];
   220→    if (arr.includes(item)) {
   221→      onChange({ [key]: arr.filter((i) => i !== item) } as Partial<SimpleForgeValues>);
   222→    } else {
   223→      onChange({ [key]: [...arr, item] } as Partial<SimpleForgeValues>);
   224→    }
   225→  }, [value, onChange]);
   226→
   227→  // I'm Lucky — escolhe combo de font aleatório (respeita locks)
   228→  const imLuckyFont = useCallback(() => {
   229→    const pool = FONT_COMBOS.filter((f) => {
   230→      // Se heading está locked, não muda heading
   231→      if (value.fontLocked.heading && f.heading !== value.fontHeading) return false;
   232→      if (value.fontLocked.body && f.body !== value.fontBody) return false;
   233→      if (value.fontLocked.mono && f.mono !== value.fontMono) return false;
   234→      return true;
   235→    });
   236→    if (pool.length === 0) {
   237→      toast.info("Todas as fonts estão bloqueadas. Desbloqueia pelo menos uma.");
   238→      return;
   239→    }
   240→    const random = pool[Math.floor(Math.random() * pool.length)];
   241→    const patch: Partial<SimpleForgeValues> = { typographyPref: "auto" };
   242→    if (!value.fontLocked.heading) patch.fontHeading = random.heading;
   243→    if (!value.fontLocked.body) patch.fontBody = random.body;
   244→    if (!value.fontLocked.mono) patch.fontMono = random.mono;
   245→    onChange(patch);
   246→    toast.success(`I'm Lucky! ${random.heading} + ${random.body} + ${random.mono}`);
   247→  }, [value.fontHeading, value.fontBody, value.fontMono, value.fontLocked, onChange]);
   248→

   253→      ? value.customColors

   255→          const trend = COLOR_TRENDS_2026.find((t) => t.id === value.colorPreset) ?? COLOR_TRENDS_2026[0];
   256→          const roles = ["Background", "Secundária", "Suporte", "Destaque"];
   257→          return trend.colors.slice(0, value.colorCount).map((hex, i) => ({ hex, role: roles[i] ?? `Cor ${i + 1}` }));
   258→        })();
   259→
   260→    // Usa a função robusta do color-engine
   261→    const polished = polishPalette(currentColors);
   262→    const effectiveTrendId = value.colorPreset !== "auto" ? value.colorPreset : "electric-lavender";
   263→    onChange({ customColors: polished, colorPreset: effectiveTrendId });
   264→    toast.success(`Polimento aplicado — tom premium ${polished.find(c => c.role === "Suporte")?.hex ?? polished[0].hex}`);
   265→  }, [value.customColors, value.colorCount, value.colorPreset, onChange]);
   266→

   268→  const generatePaletteVariation = useCallback((trendId?: string) => {
   269→    const effectiveTrendId = trendId ?? (value.colorPreset !== "auto" ? value.colorPreset : "electric-lavender");
   270→    const trend = COLOR_TRENDS_2026.find((t) => t.id === effectiveTrendId) ?? COLOR_TRENDS_2026[0];
   271→
   272→    // Usa a função robusta do color-engine
   273→    const colors = generateRandomPalette(value.colorCount, trend.colors);
   274→    onChange({ customColors: colors, colorPreset: effectiveTrendId });

   280→    if (paletteFilter === "all") return COLOR_TRENDS_2026;
   281→    return COLOR_TRENDS_2026.filter((t) => (TREND_FILTERS[t.id] ?? []).includes(paletteFilter));
   282→  }, [paletteFilter]);
   283→
   284→  // Cor ativa para preview — SEMPRE usa customColors se existirem, senão gera da tendência

   286→    const roles = ["Background", "Secundária", "Suporte", "Destaque"];

   291→      // Tem customColors mas menos que colorCount — completar com cores da tendência

   295→        result.push({ hex: trend.colors[i] ?? "#5E6AD2", role: roles[i] ?? `Cor ${i + 1}` });
   296→      }
   297→      return result;
   298→    }
   299→    const trend = COLOR_TRENDS_2026.find((t) => t.id === value.colorPreset) ?? COLOR_TRENDS_2026[0];

   301→      hex,
   302→      role: roles[i] ?? `Cor ${i + 1}`,
   303→    }));
   304→  }, [value.customColors, value.colorPreset, value.colorCount]);

   310→  const previewText = previewColors.find((c) => c.role === "Secundária")?.hex ?? "#FAFAFA";
   311→  const previewCard = previewColors.find((c) => c.role === "Suporte")?.hex ?? previewBg;
   312→  const previewMuted = previewText;

   368→      {/* Estilo Visual — com botão "Neon Auto" que liga recomendação inteligente */}
