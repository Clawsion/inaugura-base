   212→export function SimpleForge({ value, onChange, onSubmit, isLoading, onSwitchToAdvanced }: SimpleForgeProps) {
   213→  const [showExtras, setShowExtras] = useState(false);
   214→  const [showColorPopup, setShowColorPopup] = useState(false);
   215→  const [showFontPopup, setShowFontPopup] = useState(false);
   216→  const [expandedStackGroup, setExpandedStackGroup] = useState<string | null>(null);
   217→  const [showSecretMotion, setShowSecretMotion] = useState(false);
   218→  const [paletteFilter, setPaletteFilter] = useState("all");
   219→  const [colorEditIndex, setColorEditIndex] = useState<number | null>(null);
   220→
   221→  const toggleArray = useCallback((key: "references" | "mood" | "integrations", item: string) => {
   222→    const arr = value[key];
   223→    if (arr.includes(item)) {
   224→      onChange({ [key]: arr.filter((i) => i !== item) } as Partial<SimpleForgeValues>);
   225→    } else {
   226→      onChange({ [key]: [...arr, item] } as Partial<SimpleForgeValues>);
   227→    }
   228→  }, [value, onChange]);
   229→
   230→  // I'm Lucky — escolhe combo de font aleatório (respeita locks)
   231→  const imLuckyFont = useCallback(() => {
   232→    const pool = FONT_COMBOS.filter((f) => {
   233→      // Se heading está locked, não muda heading
   234→      if (value.fontLocked.heading && f.heading !== value.fontHeading) return false;
   235→      if (value.fontLocked.body && f.body !== value.fontBody) return false;
   236→      if (value.fontLocked.mono && f.mono !== value.fontMono) return false;
   237→      return true;
   238→    });
   239→    if (pool.length === 0) {
   240→      toast.info("Todas as fonts estão bloqueadas. Desbloqueia pelo menos uma.");
   241→      return;
   242→    }
   243→    const random = pool[Math.floor(Math.random() * pool.length)];
   244→    const patch: Partial<SimpleForgeValues> = { typographyPref: "auto" };
   245→    if (!value.fontLocked.heading) patch.fontHeading = random.heading;
   246→    if (!value.fontLocked.body) patch.fontBody = random.body;
   247→    if (!value.fontLocked.mono) patch.fontMono = random.mono;
   248→    onChange(patch);
   249→    toast.success(`I'm Lucky! ${random.heading} + ${random.body} + ${random.mono}`);
   250→  }, [value.fontHeading, value.fontBody, value.fontMono, value.fontLocked, onChange]);
   251→
   252→  // Polimento — dá toque moderno premium à cor (usa engine robusto)
   253→  const polishColor = useCallback(() => {
   254→    // Garantir que customColors tem as cores atuais
   255→    const currentColors: { hex: string; role: string }[] = value.customColors.length > 0
   256→      ? value.customColors
   257→      : (() => {
   258→          const trend = COLOR_TRENDS_2026.find((t) => t.id === value.colorPreset) ?? COLOR_TRENDS_2026[0];
   259→          const roles = ["Background", "Secundária", "Suporte", "Destaque"];
   260→          return trend.colors.slice(0, value.colorCount).map((hex, i) => ({ hex, role: roles[i] ?? `Cor ${i + 1}` }));
   261→        })();
   262→
   263→    // Usa a função robusta do color-engine
   264→    const polished = polishPalette(currentColors);
   265→    const effectiveTrendId = value.colorPreset !== "auto" ? value.colorPreset : "electric-lavender";
   266→    // Atualiza customColors (do trend ativo) E o override desse trend
   267→    onChange({
   268→      customColors: polished,
   269→      colorPreset: effectiveTrendId,
   270→      trendOverrides: { ...value.trendOverrides, [effectiveTrendId]: polished },
   271→    });
   272→    toast.success(`Polimento premium aplicado — tom jewel ${polished.find(c => c.role === "Suporte")?.hex ?? polished[0].hex}`);
   273→  }, [value.customColors, value.colorCount, value.colorPreset, value.trendOverrides, onChange]);
   274→
   275→  // Generate INDIVIDUAL — gera variação para UMA palete específica
   276→  // (atualiza apenas o override desse trend; se for o ativo, atualiza customColors também)
   277→  const generatePaletteVariation = useCallback((trendId?: string) => {
   278→    const effectiveTrendId = trendId ?? (value.colorPreset !== "auto" ? value.colorPreset : "electric-lavender");
   279→    const trend = COLOR_TRENDS_2026.find((t) => t.id === effectiveTrendId) ?? COLOR_TRENDS_2026[0];
   280→
   281→    // Usa a função robusta do color-engine
   282→    const colors = generateRandomPalette(value.colorCount, trend.colors);
   283→
   284→    // Atualiza o override deste trend específico
   285→    const newOverrides = { ...value.trendOverrides, [effectiveTrendId]: colors };
   286→
   287→    // Se for o trend ativo, sincroniza customColors para o preview seguir a mesma cor
   288→    if (effectiveTrendId === value.colorPreset) {
   289→      onChange({ customColors: colors, trendOverrides: newOverrides });
   290→    } else {
   291→      onChange({ trendOverrides: newOverrides });
   292→    }
   293→    toast.success(`Palete "${trend.name}" regenerada: ${value.colorCount} cores com harmonia`);
   294→  }, [value.colorCount, value.colorPreset, value.trendOverrides, onChange]);
   295→
   296→  // Generate GLOBAL — gera variação para TODAS as paletes visíveis no grid
   297→  // Cada trend recebe a sua própria combinação compatível (override independente)
   298→  const generateAllPalettes = useCallback(() => {
   299→    const newOverrides: Record<string, { hex: string; role: string }[]> = { ...value.trendOverrides };
   300→    const trendsToGenerate = paletteFilter === "all"
   301→      ? COLOR_TRENDS_2026
   302→      : COLOR_TRENDS_2026.filter((t) => (TREND_FILTERS[t.id] ?? []).includes(paletteFilter));
   303→
   304→    for (const trend of trendsToGenerate) {
   305→      newOverrides[trend.id] = generateRandomPalette(value.colorCount, trend.colors);
   306→    }
   307→
   308→    // Atualiza customColors também se houver um trend ativo
   309→    const activeTrendId = value.colorPreset !== "auto" ? value.colorPreset : null;
   310→    const patch: Partial<SimpleForgeValues> = { trendOverrides: newOverrides };
   311→    if (activeTrendId && newOverrides[activeTrendId]) {
   312→      patch.customColors = newOverrides[activeTrendId];
   313→    }
   314→    onChange(patch);
   315→    toast.success(`${trendsToGenerate.length} paletes regeneradas — todas com harmonia compatível`);
   316→  }, [value.colorCount, value.colorPreset, value.trendOverrides, paletteFilter, onChange]);
   317→
   318→  // Paletes filtradas
   319→  const filteredPalettes = useMemo(() => {
   320→    if (paletteFilter === "all") return COLOR_TRENDS_2026;
   321→    return COLOR_TRENDS_2026.filter((t) => (TREND_FILTERS[t.id] ?? []).includes(paletteFilter));
   322→  }, [paletteFilter]);
   323→
   324→  // Cor ativa para preview — SEMPRE usa customColors se existirem, senão gera da tendência
   325→  const activePalette = useMemo(() => {
   326→    const roles = ["Background", "Secundária", "Suporte", "Destaque"];
   327→    if (value.customColors.length >= value.colorCount) {
   328→      return value.customColors.slice(0, value.colorCount);
   329→    }
   330→    if (value.customColors.length > 0 && value.customColors.length < value.colorCount) {
   331→      // Tem customColors mas menos que colorCount — completar com cores da tendência
   332→      const trend = COLOR_TRENDS_2026.find((t) => t.id === value.colorPreset) ?? COLOR_TRENDS_2026[0];
   333→      const result = [...value.customColors];
   334→      for (let i = value.customColors.length; i < value.colorCount; i++) {
   335→        result.push({ hex: trend.colors[i] ?? "#5E6AD2", role: roles[i] ?? `Cor ${i + 1}` });
   336→      }
   337→      return result;
   338→    }
   339→    const trend = COLOR_TRENDS_2026.find((t) => t.id === value.colorPreset) ?? COLOR_TRENDS_2026[0];
   340→    return trend.colors.slice(0, value.colorCount).map((hex, i) => ({
   341→      hex,
   342→      role: roles[i] ?? `Cor ${i + 1}`,
   343→    }));
   344→  }, [value.customColors, value.colorPreset, value.colorCount]);
   345→
   346→  // Preview colors (sempre do tamanho colorCount) — actualizado em tempo real
   347→  const previewColors = activePalette;
   348→  const previewAccent = previewColors.find((c) => c.role === "Destaque")?.hex ?? previewColors[0]?.hex ?? "#5E6AD2";
   349→  const previewBg = previewColors.find((c) => c.role === "Background")?.hex ?? previewColors[0]?.hex ?? "#0A0A0B";
   350→  const previewText = previewColors.find((c) => c.role === "Secundária")?.hex ?? "#FAFAFA";
   351→  const previewCard = previewColors.find((c) => c.role === "Suporte")?.hex ?? previewBg;
   352→  const previewMuted = previewText;
   353→
   354→  return (
   355→    <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} className="space-y-5">
   356→      {/* Briefing */}
   357→      <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="space-y-2">
   358→        <div className="flex items-center justify-between">
   359→          <Label htmlFor="simple-briefing" className="text-sm font-semibold">Briefing do projeto <span className="text-primary">*</span></Label>
   360→          <button type="button" onClick={() => toast.info("A IA vai detetar nicho, tom, público e stack automaticamente.")} className="flex items-center gap-1 text-[10px] text-primary hover:underline">
   361→            <Lightbulb className="h-3 w-3" /> Auto-detetar a partir do briefing
   362→          </button>
   363→        </div>
   364→        <Textarea id="simple-briefing" value={value.briefing} onChange={(e) => onChange({ briefing: e.target.value })}
   365→          placeholder="Ex: Estou a criar uma plataforma SaaS B2B para gestão de equipas remotas. O público-alvo são CTOs e Head of Ops de startups em fase Series A-B..."
   366→          className="min-h-[140px] resize-y border-border bg-card/50 text-sm leading-relaxed" />
   367→        <p className="text-xs text-muted-foreground">{value.briefing.length} caracteres · mínimo 20</p>
   368→      </motion.div>
   369→
   370→      {/* Tipo de Negócio / Nicho — por baixo do briefing */}
   371→      <div className="space-y-1.5">
   372→        <Label className="text-xs font-medium text-muted-foreground">Tipo de Negócio / Nicho</Label>
   373→        <input type="text" value={value.nicho} onChange={(e) => onChange({ nicho: e.target.value })}
   374→          placeholder="Ex: SaaS B2B, E-commerce moda, Restaurante, Agência criativa..."
   375→          className="w-full rounded-lg border border-border bg-card/50 px-3 py-2 text-sm" />
